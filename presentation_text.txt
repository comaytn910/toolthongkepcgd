let currentSlide = 1;
        const totalSlides = 10;

        // Khai báo các phần tử DOM quan trọng
        const slides = document.querySelectorAll('.slide-container');
        const dots = document.querySelectorAll('.slide-dot');
        const indicator = document.getElementById('slideIndicator');
        const btnPrev = document.getElementById('btnPrev');
        const btnNext = document.getElementById('btnNext');
        const btnFullscreen = document.getElementById('btnFullscreen');
        const slideDeck = document.getElementById('slideDeck');

        // Hàm chuyển đổi sang slide mong muốn
        function goToSlide(slideNum) {
            if(slideNum < 1 || slideNum > totalSlides) return;
            
            // Ẩn tất cả slides cũ
            slides.forEach(slide => {
                slide.classList.remove('slide-active');
                slide.classList.add('slide-hidden');
            });

            // Hiển thị slide mới
            const targetSlide = document.getElementById('slide' + slideNum);
            if(targetSlide) {
                targetSlide.classList.remove('slide-hidden');
                targetSlide.classList.add('slide-active');
            }

            // Cập nhật trạng thái các nút chấm tròn định vị
            dots.forEach((dot, idx) => {
                if(idx + 1 === slideNum) {
                    dot.classList.remove('bg-slate-700');
                    dot.classList.add('bg-teal-500', 'w-6');
                } else {
                    dot.classList.remove('bg-teal-500', 'w-6');
                    dot.classList.add('bg-slate-700');
                }
            });

            currentSlide = slideNum;
            indicator.innerText = `Slide ${currentSlide} / ${totalSlides}`;
            
            // Bật/tắt nút điều hướng khi ở slide đầu/cuối
            btnPrev.disabled = currentSlide === 1;
            btnPrev.style.opacity = currentSlide === 1 ? '0.4' : '1';
            btnNext.disabled = currentSlide === totalSlides;
            btnNext.style.opacity = currentSlide === totalSlides ? '0.4' : '1';
        }

        // Sự kiện click nút bấm điều hướng
        btnPrev.addEventListener('click', () => goToSlide(currentSlide - 1));
        btnNext.addEventListener('click', () => goToSlide(currentSlide + 1));

        // Hỗ trợ bấm phím mũi tên bàn phím cực kỳ trực quan
        document.addEventListener('keydown', (e) => {
            if(e.key === 'ArrowRight' || e.key === 'Space') {
                e.preventDefault();
                goToSlide(currentSlide + 1);
            } else if(e.key === 'ArrowLeft') {
                e.preventDefault();
                goToSlide(currentSlide - 1);
            }
        });

        // Bật/tắt chế độ xem toàn màn hình
        btnFullscreen.addEventListener('click', () => {
            if (!document.fullscreenElement) {
                slideDeck.requestFullscreen().catch(err => {
                    console.error(`Lỗi toàn màn hình: ${err.message}`);
                });
            } else {
                document.exitFullscreen();
            }
        });

        // Hàm zoom ảnh thông minh tại Slide 9
        function zoomInfographic(level, btnElement) {
            const viewport = document.getElementById('imgViewport');
            const buttons = document.querySelectorAll('.zoom-opt-btn');
            
            // Xóa tất cả hiệu ứng zoom cũ
            viewport.classList.remove('zoom-all', 'zoom-top', 'zoom-middle', 'zoom-bottom');
            
            // Áp dụng lớp phóng to mới
            viewport.classList.add('zoom-' + level);
            
            // Đặt lại phong cách cho toàn bộ nút lựa chọn
            buttons.forEach(btn => {
                btn.classList.remove('bg-teal-500', 'text-slate-950', 'font-bold', 'border-teal-400');
                btn.classList.add('bg-slate-900', 'text-slate-300', 'font-semibold', 'border-slate-800');
            });
            
            // Làm nổi bật nút đang được kích hoạt
            if (btnElement) {
                btnElement.classList.remove('bg-slate-900', 'text-slate-300', 'font-semibold', 'border-slate-800');
                btnElement.classList.add('bg-teal-500', 'text-slate-950', 'font-bold', 'border-teal-400');
            }
        }

        // Khởi động hiển thị slide ban đầu
        window.onload = function() {
            goToSlide(1);
        };
